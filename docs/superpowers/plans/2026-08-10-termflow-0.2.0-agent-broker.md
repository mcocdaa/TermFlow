# TermFlow 0.2.0: Agent Broker Architecture — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-subagent-driven-development (recommended) or superpowers-executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade the TermFlow control plane (B) to a Terminal Capability Broker + Event Broker + Agent Gateway, enabling external AI agents (opencode) to read/write remote panes via MCP tools, watch for terminal events, and surface agent interactions in a new C-side AgentChat UI panel.

**Architecture:** MCP server at `/api/v1/mcp` exposes 8 tools gated by agent-token auth. A persistent Watch Engine in B tracks pane conditions and delivers wake-ups to opencode via HTTP. An OpenCodeAdapter bridges B's AgentDriver port to an opencode Docker container. The C web/Tauri client gets a floating AgentChat panel connected via a new WebSocket stream.

**Tech Stack:** Python 3.12 (uv workspace), FastAPI, Pydantic, SQLite/aiosqlite, asyncio. TypeScript (Vue 3) for C UI. MCP 2025-06-18 Streamable HTTP. opencode serve HTTP API.

```python
async def _handle_snapshot_request(self, request: SnapshotRequestPayload) -> None:
    topology = self.topology_provider()
    pane = next((p for w in topology.windows for p in w.panes if p.pane_id == request.pane_id), None)
    if pane is None:
        result = SnapshotResultPayload(command_id=request.command_id, ok=False, error_code="pane_not_found")
    else:
        try:
            raw = await self.control.capture_pane(request.pane_id)
            chunk = self.buffers.append(request.pane_id, raw)
            result = SnapshotResultPayload(
                command_id=request.command_id, ok=True,
                data_base64=base64.b64encode(raw).decode("ascii"),
                stream_id=chunk.stream_id, seq=chunk.seq,
                rows=pane.height, cols=pane.width,
            )
        except Exception:
            result = SnapshotResultPayload(command_id=request.command_id, ok=False, error_code="snapshot_failed")
    self.transport.enqueue_nowait(
        self._message(MessageType.PANE_SNAPSHOT_RESULT, result.model_dump(mode="json"))
    )
```

```python
"""ANSI escape stripping and OSC133 shell-marker parsing."""
import re
from dataclasses import dataclass, field
from enum import Enum

class Osc133Marker(Enum):
    A = "A"  # prompt ready
    B = "B"  # command started
    C = "C"  # output started
    D = "D"  # command finished

@dataclass(frozen=True, slots=True)
class Osc133Event:
    marker: Osc133Marker
    exit_code: int | None = None

_ANSI_ESCAPE = re.compile(r"\x1b\[[0-?]*[ -/]*[@-~]")
_ANSI_OSC = re.compile(r"\x1b\][^\x07\x1b]*(?:\x07|\x1b\\)")

def strip_ansi(text: str) -> str:
    text = _ANSI_ESCAPE.sub("", text)
    text = _ANSI_OSC.sub("", text)
    return text

_OSC133_RE = re.compile(r"\x1b\]133;([ABCD]);?[^\x07\x1b]*(?:\x07|\x1b\\)")

def has_command_finished(text: str) -> bool:
    return bool(_OSC133_RE.search(text) and "133;D" in text)

def has_prompt_ready(text: str) -> bool:
    return bool(_OSC133_RE.search(text) and "133;A" in text)
```

```python
router = APIRouter(prefix="/api/v1/mcp", tags=["mcp"])

@router.post("")
async def mcp_streamable_http(
    request: Request,
    binding: Annotated[AgentBindingModel, Depends(get_agent_binding)],
    repositories: Annotated[RepositoryBundle, Depends(get_repositories)],
    registry: Annotated[LiveInstanceRegistry, Depends(get_registry)],
    cmd_router: Annotated[CommandRouter, Depends(get_command_router)],
) -> JSONResponse:
    """Handles MCP initialize, tools/list, tools/call, notifications/initialized."""
    body = await request.json()
    method = body.get("method", "")
    request_id = body.get("id")

    if method == "initialize":
        return JSONResponse({"jsonrpc":"2.0","id":request_id,"result":{
            "protocolVersion":"2025-06-18",
            "serverInfo":{"name":"termflow","version":"0.2.0"},
            "capabilities":{"tools":{}},
        }})
    if method == "tools/list":
        return JSONResponse({"jsonrpc":"2.0","id":request_id,"result":{"tools": _MCP_TOOLS}})
    if method == "tools/call":
        tool_name = body.get("params",{}).get("name","")
        tool_args = body.get("params",{}).get("arguments",{})
        pane_cache = cast(PaneCache, request.app.state.pane_cache)
        result = await _dispatch_tool(tool_name, tool_args,
            binding=binding, repositories=repositories, registry=registry,
            command_router=cmd_router, pane_cache=pane_cache)
        return JSONResponse({"jsonrpc":"2.0","id":request_id,"result":{
            "content":[{"type":"text","text":json.dumps(result,default=str)}]
        }})
    if method == "notifications/initialized":
        return JSONResponse({"jsonrpc":"2.0","id":request_id,"result":{}})
    return JSONResponse({"jsonrpc":"2.0","id":request_id,
        "error":{"code":-32601,"message":"Method not found"}}, status_code=404)
```

```python
async def get_agent_binding(
    request: Request,
    repositories: Annotated[RepositoryBundle, Depends(get_repositories)],
) -> object:
    """Extract agent token from Bearer auth, resolve to AgentBinding."""
    raw = request.headers.get("authorization", "")
    scheme, _, token = raw.partition(" ")
    if scheme.lower() != "bearer" or not token:
        raise TermFlowError("unauthorized", 401, "Agent token required.")
```

```python
class WatchEngine:
    def __init__(self, repositories, pane_cache, check_interval_seconds=1.0):
        self._repos = repositories
        self._pane_cache = pane_cache
        self._check_interval = check_interval_seconds
        self._last_pane_output: dict[str, datetime] = {}  # pane_id -> last output time
        self._triggered_queue: asyncio.Queue[WatchModel] = asyncio.Queue()

    def record_pane_output(self, pane_id: str) -> None:
        """Record that the pane produced output (for silent_for checks)."""
        self._last_pane_output[pane_id] = datetime.now(UTC)

    async def scan(self, active_watches: list[WatchModel]) -> list[WatchModel]:
        """Evaluate active watches, return triggered ones."""
        triggered: list[WatchModel] = []
        now = datetime.now(UTC)
        for watch in active_watches:
            if watch.state != "active":
                continue
            if await self._evaluate_watch(watch):
                watch.state = "triggered"
                watch.triggered_at = now
                await self._repos.watches.update_state(watch.id, "triggered", triggered_at=now)
                await self._triggered_queue.put(watch)
                triggered.append(watch)
        return triggered

    async def _evaluate_watch(self, watch: WatchModel) -> bool:
        cond = json.loads(watch.condition)
        cond_type = cond.get("type", "")
        if cond_type == "silent_for":
            return self._eval_silent_for(watch, int(cond.get("value", 0)))
        if cond_type == "output_match":
            return await self._eval_output_match(watch, str(cond.get("value", "")))
        if cond_type == "command_finished":
            entry = await self._pane_cache.get(watch.pane_id)
            return entry is not None and has_command_finished(entry.text)
        if cond_type == "prompt_ready":
            entry = await self._pane_cache.get(watch.pane_id)
            return entry is not None and has_prompt_ready(entry.text)
        # pane_exited is checked externally via topology scan
        return False

    def _eval_silent_for(self, watch: WatchModel, seconds: int) -> bool:
        last = self._last_pane_output.get(watch.pane_id)
        if last is None:
            return False
        return (datetime.now(UTC) - last).total_seconds() >= seconds

    async def _eval_output_match(self, watch: WatchModel, pattern: str) -> bool:
        entry = await self._pane_cache.get(watch.pane_id)
        if entry is None or not entry.text:
            return False
        try:
            return bool(re.compile(pattern).search(entry.text))
        except re.error:
            return False

    async def triggered(self) -> WatchModel:
        """Await the next triggered watch from the queue."""
        return await self._triggered_queue.get()
```

```python
class OpenCodeAdapter(AgentDriver):
    def __init__(self, settings):
        self._base_url = settings.opencode_server_url.rstrip("/")
        pwd = settings.opencode_server_password.get_secret_value()
        self._auth = {"Authorization": "Basic " + base64.b64encode(f"opencode:{pwd}".encode()).decode()}
        self._client: httpx.AsyncClient | None = None

    async def _ensure_client(self) -> httpx.AsyncClient:
        if self._client is None:
            self._client = httpx.AsyncClient(timeout=httpx.Timeout(30.0))
        return self._client

    async def create_session(self, binding_id) -> str:
        client = await self._ensure_client()
        resp = await client.post(f"{self._base_url}/session", headers=self._auth,
                                 json={"binding_id": str(binding_id)})
        resp.raise_for_status()
        return str(resp.json()["session_id"])

    async def send_message(self, session_id, envelope: AgentInputEnvelope):
        client = await self._ensure_client()
        if envelope.kind == "user_message":
            prompt = envelope.payload.text
        elif envelope.kind == "watch_triggered":
            wt = envelope.payload  # type: ignore[union-attr]
            prompt = (
                f"[TermFlow Watch {wt.watch_id} triggered]\n"
                f"pane: {wt.pane_id}   condition: {wt.condition}\n"
                f"note: {wt.note or '(none)'}\n"
                f"latest output:\n{wt.latest_output or '(no output)'}"
            )
        else:
            return
        resp = await client.post(
            f"{self._base_url}/session/{session_id}/message",
            headers=self._auth,
            json={"prompt": prompt, "kind": envelope.kind},
        )
        resp.raise_for_status()

    async def subscribe_events(self, session_id) -> AsyncIterator[AgentEvent]:
        client = await self._ensure_client()
        async with client.stream("GET", f"{self._base_url}/session/{session_id}/events",
                                 headers=self._auth) as resp:
            resp.raise_for_status()
            async for line in resp.aiter_lines():
                if line.startswith("data: "):
                    event = self._parse_opencode_event(json.loads(line[6:]), session_id)
                    if event:
                        yield event

    async def abort(self, session_id):
        client = await self._ensure_client()
        resp = await client.post(f"{self._base_url}/session/{session_id}/abort", headers=self._auth)
        resp.raise_for_status()

    async def health(self) -> bool:
        try:
            client = await self._ensure_client()
            resp = await client.get(f"{self._base_url}/global/health", headers=self._auth)
            return resp.status_code == 200
        except httpx.HTTPError:
            return False
```

```python
class WriteLeaseManager:
    """In-memory per-pane write lease tracker. Safe direction on restart."""
    def __init__(self):
        self._leases: dict[str, WriteLease] = {}
        self._lock = asyncio.Lock()

    async def get_or_create(self, pane_id: str) -> WriteLease:
        async with self._lock:
            if pane_id not in self._leases:

            return self._leases[pane_id]

        return lease.increment()

    async def check(self, pane_id: str, expected: int) -> bool:
        lease = await self.get_or_create(pane_id)
        return lease.check(expected)

    async def get_generation(self, pane_id: str) -> int:
        lease = await self.get_or_create(pane_id)
        return lease.generation
```

```typescript
export function useAgentSession(bindingId: string) {
  const runtime = useClientRuntime()
  const connected = ref(false)
  const messages = ref<AgentFrameIn[]>([])
  const streamingContent = ref('')
  const isStreaming = ref(false)
  const error = ref<string | null>(null)
  let ws: WebSocket | null = null
  let reconnectTimer: unknown = null

  function connect() {
    const url = new URL('/api/v1/agent/stream', runtime.canonicalServerUrl)
    url.protocol = url.protocol === 'https:' ? 'wss:' : 'ws:'
    url.searchParams.set('binding_id', bindingId)
    ws = new WebSocket(url.toString())

    ws.onopen = () => { connected.value = true; error.value = null }
    ws.onmessage = (event: MessageEvent<string>) => {
      try {
        const frame: AgentFrameIn = JSON.parse(event.data)
        if (frame.type === 'agent.message.delta') {
          streamingContent.value += frame.content
          isStreaming.value = true
        } else if (frame.type === 'agent.run.finished') {
          if (streamingContent.value) {
            messages.value.push({ type: 'agent.message.delta' as const, binding_id: bindingId, content: streamingContent.value })
            streamingContent.value = ''
          }
          isStreaming.value = false
        }
        messages.value.push(frame)
      } catch { /* ignore malformed */ }
    }
    ws.onclose = (e) => {
      connected.value = false
      if (e.code !== 1000 && e.code !== 4401) {
        reconnectTimer = runtime.clock.setTimeout(() => { if (!connected.value) connect() }, 3000)
      }
    }
    ws.onerror = () => { error.value = 'Agent connection error' }
  }

  function sendMessage(text: string) {
    if (!ws || ws.readyState !== WebSocket.OPEN) return
    ws.send(JSON.stringify({ type: 'agent.message.send', binding_id: bindingId, text }))
  }
  function sendPermissionResponse(requestId: string, approved: boolean) {
    if (!ws || ws.readyState !== WebSocket.OPEN) return
    ws.send(JSON.stringify({ type: 'agent.permission.response', binding_id: bindingId, request_id: requestId, approved }))
  }
  function disconnect() {
    if (reconnectTimer !== null) { runtime.clock.clearTimeout(reconnectTimer); reconnectTimer = null }
    if (ws) { ws.close(1000); ws = null }
    connected.value = false
  }
  onBeforeUnmount(disconnect)

  return { connected, messages, streamingContent, isStreaming, error, connect, sendMessage, sendPermissionResponse, disconnect }
}
```

```python
class TestAgentBrokerIntegration:
    def test_list_panes_through_mcp(self, client, agent_headers):
        """Create a binding, get token, call termflow_list_panes"""
        resp = client.post("/api/v1/mcp", headers=agent_headers, json={
            "jsonrpc":"2.0","id":1,"method":"tools/call",
            "params":{"name":"termflow_list_panes","arguments":{}}
        })
        assert resp.status_code == 200
        content = json.loads(resp.json()["result"]["content"][0]["text"])
        assert "panes" in content

    def test_watch_lifecycle(self, client, agent_headers):
        """Create watch → list watches → cancel watch"""
        # create
        resp = client.post("/api/v1/mcp", headers=agent_headers, json={
            "jsonrpc":"2.0","id":1,"method":"tools/call",
            "params":{"name":"termflow_watch_create","arguments":{"pane_id":"%1","condition":"pane_exited"}}
        })
        data = json.loads(resp.json()["result"]["content"][0]["text"])
        watch_id = data["watch_id"]
        # list
        resp = client.post("/api/v1/mcp", headers=agent_headers, json={
            "jsonrpc":"2.0","id":2,"method":"tools/call",
            "params":{"name":"termflow_watch_list","arguments":{}}
        })
        watches = json.loads(resp.json()["result"]["content"][0]["text"])["watches"]
        assert any(w["watch_id"] == watch_id for w in watches)
        # cancel
        resp = client.post("/api/v1/mcp", headers=agent_headers, json={
            "jsonrpc":"2.0","id":3,"method":"tools/call",
            "params":{"name":"termflow_watch_cancel","arguments":{"watch_id":watch_id}}
        })
        assert json.loads(resp.json()["result"]["content"][0]["text"])["cancelled"] is True
```
